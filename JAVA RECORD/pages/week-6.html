☕ StudentCRUD.java
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.ResultSet;
import java.sql.Statement;

public class StudentCRUD {
    public static void main(String[] args) {
        try {
            Class.forName("com.mysql.cj.jdbc.Driver");
            Connection con = DriverManager.getConnection(
                "jdbc:mysql://localhost:3306/it", "root", ""
            );
            Statement st = con.createStatement();

            // CREATE (Insert)
            st.executeUpdate(
                "INSERT INTO Student VALUES(101, 'Rahul', 'CSE', 90)"
            );
            System.out.println("Record Inserted Successfully");

            // READ (Select)
            ResultSet rs = st.executeQuery("SELECT * FROM Student");
            System.out.println("\nStudent Records:");
            while (rs.next()) {
                System.out.println(
                    rs.getInt(1) + "  " +
                    rs.getString(2) + "  " +
                    rs.getString(3) + "  " +
                    rs.getInt(4)
                );
            }

            // UPDATE
            st.executeUpdate(
                "UPDATE Student SET marks = 95 WHERE id = 101"
            );
            System.out.println("\nRecord Updated Successfully");

            // DELETE
            st.executeUpdate(
                "DELETE FROM Student WHERE id = 101"
            );
            System.out.println("\nRecord Deleted Successfully");

            con.close();
        } catch (Exception e) {
            System.out.println(e);
        }
    }
}

📊 6. Result Analysis & Discussion

🔍 Statement vs PreparedStatement

While raw Statement objects were utilized for simple demonstration queries, production CRUD applications must migrate to PreparedStatement to leverage precompiled database execution plans and eliminate SQL injection vulnerabilities.

⚡ Cursor Navigation Performance

Forward-only ResultSet cursors (TYPE_FORWARD_ONLY) maintain a minimal memory footprint on the client machine by streaming database rows sequentially, outperforming bidirectional scrollable cursors on large datasets.

🛡️ Relational Schema Constraints

The id INT PRIMARY KEY constraint enforces entity integrity in MySQL, immediately rejecting duplicate key insertions with a MySQLIntegrityConstraintViolationException.

💡 DML Transaction Commit Verification

Each call to st.executeUpdate() atomically alters the underlying database state, verified via corresponding MySQL CLI queries confirming real-time persistence.
