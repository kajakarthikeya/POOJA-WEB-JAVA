package javaapplication26;

import java.sql.*;

public class BookCRUD {
    public static void main(String[] args) {
        try {
            Class.forName("com.mysql.jdbc.Driver");
            Connection con = DriverManager.getConnection("jdbc:mysql://localhost/it", "root", "");
            Statement st = con.createStatement();

            // INSERT
            st.executeUpdate("INSERT INTO Book VALUES(101,'Java',4500.0)");
            System.out.println("Book Inserted Successfully\n");
            System.out.println("Book Details");
            ResultSet rs = st.executeQuery("SELECT * FROM Book");
            while (rs.next()) {
                System.out.println(rs.getInt(1) + " " + rs.getString(2) + " " + rs.getDouble(3));
            }

            // UPDATE
            st.executeUpdate("UPDATE Book SET Price=5000.0 WHERE BookNo=101");
            System.out.println("\nBook Updated Successfully");
            rs = st.executeQuery("SELECT * FROM Book");
            while (rs.next()) {
                System.out.println(rs.getInt(1) + " " + rs.getString(2) + " " + rs.getDouble(3));
            }

            // DELETE
            st.executeUpdate("DELETE FROM Book WHERE BookNo=101");
            System.out.println("\nBook Deleted Successfully");
            con.close();
        } catch (Exception e) {
            System.out.println(e);
        }
    }
}
